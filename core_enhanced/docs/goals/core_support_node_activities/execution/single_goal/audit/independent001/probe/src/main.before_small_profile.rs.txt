//! Audit-only public commands; no private state or save edits.
use game_core::{application::{ApplicationCommand, ApplicationSnapshotKind, CoreApplication}, game::data::content_pack, runtime_observation::begin_lab_window, runtime_safety::RuntimeSafetySettings};
use serde_json::{json, Value};
use std::{fs, path::Path, sync::Arc};
use uuid::Uuid;
const PLAYER: Uuid = Uuid::from_u128(0xc0de1200000000000000000000001001);
fn snapshot(app: &CoreApplication) -> Value {
    app.snapshot(ApplicationSnapshotKind::Run).unwrap().read_model().as_value().clone()
}
fn main() {
    let args: Vec<String> = std::env::args().skip(1).collect();
    let config: Value = serde_json::from_slice(&fs::read(&args[0]).unwrap()).unwrap();
    let replay: Value = serde_json::from_slice(&fs::read(&args[1]).unwrap()).unwrap();
    let selection = serde_json::from_slice(&fs::read(config["base_selection_path"].as_str().unwrap()).unwrap()).unwrap();
    let original = fs::read_to_string(config["runtime_safety_path"].as_str().unwrap()).unwrap();
    let mut rows = Vec::new();
    for limit in [16777216, 1024] {
        let settings_text = original.replacen("max_retained_bytes: 16777216", &format!("max_retained_bytes: {limit}"), 1);
        let settings = Arc::new(RuntimeSafetySettings::from_ron("audit trusted smaller host setting", settings_text.as_bytes()).unwrap());
        let data = content_pack::load_with_runtime_safety(Path::new(config["content_root"].as_str().unwrap()), &selection, &content_pack::InputLimits::default(), settings).unwrap();
        let mut app = CoreApplication::new(data, 12077);
        for (i, command) in replay["commands"].as_array().unwrap().iter().enumerate() {
            app.execute(PLAYER, ApplicationCommand::from_document(command.clone().into()).unwrap(), Default::default()).unwrap_or_else(|e| panic!("setup {i}: {e:?}"));
        }
        let before = snapshot(&app);
        let offer = before["selected_event"]["offers"].as_array().unwrap().iter().find(|o| o["merchandise"]["type"] == "relic").expect("relic offer").clone();
        let window = begin_lab_window().unwrap();
        let first = app.execute(PLAYER, ApplicationCommand::from_document(json!({"type":"purchase_item", "offer_id":offer["offer_id"]}).into()).unwrap(), Default::default());
        let first_observation = window.finish().unwrap();
        let after = snapshot(&app);
        let window = begin_lab_window().unwrap();
        let following = app.execute(PLAYER, ApplicationCommand::from_document(json!({"type":"request_relic_state"}).into()).unwrap(), Default::default());
        let following_observation = window.finish().unwrap();
        rows.push(json!({"limit":limit,"offer":offer,"before_money":before["resources"]["enkephalin"],"after_money":after["resources"]["enkephalin"],"purchase_ok":first.is_ok(),"purchase_result":first.as_ref().ok().map(|v|v.payload().as_value()),"purchase_error":first.as_ref().err().map(|e|format!("{e:?}")),"purchase_observation":first_observation,"following_ok":following.is_ok(),"following_error":following.as_ref().err().map(|e|format!("{e:?}")),"following_observation":following_observation}));
    }
    fs::write(&args[2], serde_json::to_vec_pretty(&json!({"scope":"audit-only public CoreApplication; supported content and trusted runtime profile; no save/state mutation", "rows":rows})).unwrap()).unwrap();
}
