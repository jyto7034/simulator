result['rust']={'unique_core_passed':len(passed),'suites':{s:len(names(E/f'runs/{s}.log','ok')) for s in suites},'closed_original_failures':sorted(failures),'existing_ignored':6,'native_reused':4,'server_reused':31,'single_green_invocation':False,'coverage_selector_checks':selectors}
